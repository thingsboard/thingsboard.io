export const pipPackage = (product) =>
    product === Products.CE
        ? 'tb-ce-client'
        : product === Products.PE
            ? 'tb-pe-client'
            : 'tb-paas-client';

export const importModule = (product) => pipPackage(product).replace(/-/g, '_');

export const pipInstall = (product) => `pip install ${pipPackage(product)}==${PYTHON_CLIENT_VER}`;

The ThingsBoard Python client wraps the <DocLink product={props.product} path='reference/rest-api'>REST API</DocLink> and is auto-generated from the OpenAPI spec. Edition-specific packages ensure type-safe access to the right endpoints for your tenant.

<Aside type="tip">
Source and additional examples are available in the [thingsboard-python-client](https://github.com/thingsboard/thingsboard-python-client) repository.
For a Java alternative, see the <DocLink product={props.product} path='reference/java-client'>Java client</DocLink>.
</Aside>

## Authentication

### Username and password (JWT)

Authenticates at construction time via `/api/auth/login`. The client refreshes the JWT automatically before it expires.

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient

client = ThingsboardClient(
    "{BASE_URL}",
    username="tenant@thingsboard.org",
    password="tenant",
)

print(client.get_user())`} />

### API key

Sets `X-Authorization: ApiKey <key>` on every request; no login call is made.

<Aside type="tip">
Available since ThingsBoard 4.3. See the <DocLink product={props.product} path='user-guide/security/api-keys'>API Keys</DocLink> guide.
</Aside>

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient

client = ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE")

print(client.get_user())`} />

### Pre-existing token

Inject an externally obtained JWT pair when your application already manages tokens itself.

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient

client = ThingsboardClient(
    "{BASE_URL}",
    token="eyJhbGciOi...",
    refresh_token="eyJhbGciOi...",
)`} />

## Resource cleanup

Use the client as a context manager so the underlying connection pool is released automatically on exit:

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient

with ThingsboardClient(
    "{BASE_URL}",
    username="tenant@thingsboard.org",
    password="tenant",
) as client:
    devices = client.get_tenant_devices(page_size=10, page=0)
    for device in devices.data:
        print(device.name, device.id.get_id())`} />

## Manage devices

Create a device, fetch it by ID, then delete it:

<HostCode product={props.product} lang="python" meta="maxLines=25 collapsible" code={`from ${importModule(props.product)} import ThingsboardClient
from ${importModule(props.product)}.models import Device

with ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE") as client:
    new_device = Device(name="Test Device", type="default")
    saved_device = client.save_device(new_device)
    print("Saved device:", saved_device)

    device_id = saved_device.id.get_id()
    found_device = client.get_device_by_id(device_id=device_id)
    print("Found device:", found_device)

    client.delete_device(device_id=device_id)`} />

## Fetch tenant devices

Paginate through every device on the tenant:

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient

with ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE") as client:
    page = 0
    while True:
        tenant_devices = client.get_tenant_devices(page_size=10, page=page)
        for device in tenant_devices.data:
            print(device.name, device.id.get_id())
        if not tenant_devices.has_next:
            break
        page += 1`} />

## Manage attributes (read-modify-write)

Read an attribute value, change it, and write it back:

<HostCode product={props.product} lang="python" meta="maxLines=25 collapsible" code={`import json
from ${importModule(props.product)} import ThingsboardClient

with ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE") as client:
    asset_id = "YOUR_ASSET_ID"

    attrs = client.get_attributes(
        entity_type="ASSET",
        entity_id=asset_id,
        keys="deviceCount",
    )

    current = int(attrs[0].value) if attrs else 0
    updated = current + 1

    client.save_entity_attributes_v2(
        entity_type="ASSET",
        entity_id=asset_id,
        scope="SERVER_SCOPE",
        body=json.dumps({"deviceCount": updated}),
    )`} />

## Push telemetry

Send time-series telemetry for a device. The `body` parameter accepts a JSON string, not a Python dict:

<HostCode product={props.product} lang="python" code={`import json
from ${importModule(props.product)} import ThingsboardClient

with ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE") as client:
    device_id = "YOUR_DEVICE_ID"
    client.save_entity_telemetry(
        entity_type="DEVICE",
        entity_id=device_id,
        scope="ANY",
        body=json.dumps({"temperature": 26.5, "humidity": 87}),
    )`} />

## Error handling

The client raises `ApiException` for any non-2xx response and `NotFoundException` (a subclass of `ApiException`) for 404s. Catch the specific subclass when you only care about missing entities:

<HostCode product={props.product} lang="python" code={`from ${importModule(props.product)} import ThingsboardClient
from ${importModule(props.product)}.exceptions import ApiException, NotFoundException

with ThingsboardClient("{BASE_URL}", api_key="YOUR_API_KEY_VALUE") as client:
    try:
        device = client.get_device_by_id(device_id="nonexistent-id")
    except NotFoundException:
        print("Device not found")
    except ApiException as e:
        print(f"API error {e.status}: {e.body}")`} />
